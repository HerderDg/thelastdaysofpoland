

set_initial_team = {
	log = "---------------------------------------------------"
	set_variable = {
		next_tee_number = 13
	}
	every_character = {
		limit = {
			has_character_flag = is_footballer
			has_character_flag = is_active
		}
		clr_character_flag = is_active
	}
	set_temp_variable = {
		attackers = 0
	}
	while_loop_effect = {
		limit = {
			check_variable = {
				attackers < 2
			}
		}
		set_temp_variable = {
			max_power = 0
		}
		every_character = {
			limit = {
				has_character_flag = is_footballer
				has_character_flag = is_attacker
				NOT = {
					has_character_flag = is_active
				}
			}
			log = "current max power: [?max_power]"
			log = "current char: [THIS.GetName] ([?THIS.fb_power])"
			IF = {
				limit = {
					check_variable = {
						max_power = 0
					}
				}
				set_temp_variable = {
					max_power = THIS.fb_power
				}
				set_character_flag = initially_chosen
			}
			ELSE_IF = {
				limit = {
					check_variable = {
						max_power < THIS.fb_power
					}
				}
				log = "change chosen char to [THIS.GetName]"
				set_temp_variable = {
					max_power = THIS.fb_power
				}
				ROOT = {
					every_character = {
						limit = {
							has_character_flag = initially_chosen
						}
						clr_character_flag = initially_chosen
						log = "[THIS.GetName] loses the flag"
					}
				}
				set_character_flag = initially_chosen
			}
		}
		every_character = {
			limit = {
				has_character_flag = initially_chosen
			}
			clr_character_flag = initially_chosen
			set_character_flag = is_active
			log = "[THIS.GetName] was chosen to team"
		}
		add_to_temp_variable = {
			attackers = 1
		}
		log = "loop ended"
	}
	every_character = {
		limit = {
			has_character_flag = is_footballer
			NOT = {
				has_character_flag = is_attacker
			}
		}
		set_character_flag = is_active
	}
	random_character = {
		limit = {
			has_character_flag = is_attacker
			has_character_flag = is_active
		}
		set_character_flag = left_atk
		set_variable = {ROOT.left_atk_tee_number = THIS.tee_number}
		set_variable = {ROOT.left_atk_name = THIS}
		set_variable = {ROOT.left_atk_fb = THIS.fb_power}
		log = "[THIS.GetName] is on left attack"
	}
	log = "L: [?left_atk_name.GetName]"
	every_character = {
		limit = {
			has_character_flag = is_attacker
			has_character_flag = is_active
			NOT = {
				has_character_flag = left_atk
			}
		}
		log = ">[THIS.GetName]"
	}
	random_character = {
		limit = {
			has_character_flag = is_attacker
			has_character_flag = is_active
			NOT = {
				has_character_flag = left_atk
			}
		}
		set_character_flag = right_atk
		set_variable = {ROOT.right_atk_tee_number = THIS.tee_number}
		set_variable = {ROOT.right_atk_name = THIS}
		set_variable = {ROOT.right_atk_fb = THIS.fb_power}
		log = "[THIS.GetName] is on right attack"
	}
	log = "R: [?right_atk_name.GetName]"
	random_character = {
		limit = {
			has_character_flag = is_midfielder
			has_character_flag = is_active
		}
		set_character_flag = left_mid
		set_variable = {ROOT.left_mid_tee_number = THIS.tee_number}
		set_variable = {ROOT.left_mid_name = THIS}
		set_variable = {ROOT.left_mid_fb = THIS.fb_power}
		log = "[THIS.GetName] is on left mid"
	}
	random_character = {
		limit = {
			has_character_flag = is_midfielder
			has_character_flag = is_active
			NOT = {
				has_character_flag = left_mid
			}
		}
		set_character_flag = left_mid
		set_character_flag = center_mid
		set_variable = {ROOT.left_center_mid_tee_number = THIS.tee_number}
		set_variable = {ROOT.left_center_mid_name = THIS}
		set_variable = {ROOT.left_center_mid_fb = THIS.fb_power}
		log = "[THIS.GetName] is on left-center mid"
	}
	random_character = {
		limit = {
			has_character_flag = is_midfielder
			has_character_flag = is_active
			NOT = {
				has_character_flag = left_mid
			}
		}
		set_character_flag = right_mid
		set_character_flag = center_mid
		set_variable = {ROOT.right_center_mid_tee_number = THIS.tee_number}
		set_variable = {ROOT.right_center_mid_name = THIS}
		set_variable = {ROOT.right_center_mid_fb = THIS.fb_power}
		log = "[THIS.GetName] is on right-center mid"
	}
	random_character = {
		limit = {
			has_character_flag = is_midfielder
			has_character_flag = is_active
			NOT = {
				has_character_flag = left_mid
			}
			NOT = {
				has_character_flag = center_mid
			}
			NOT = {
				has_character_flag = right_mid
			}
		}
		set_character_flag = right_mid
		set_variable = {ROOT.right_mid_tee_number = THIS.tee_number}
		set_variable = {ROOT.right_mid_name = THIS}
		set_variable = {ROOT.right_mid_fb = THIS.fb_power}
		log = "[THIS.GetName] is on right mid"
	}
	random_character = {

			has_character_flag = is_defender
			has_character_flag = is_active
		}
		set_character_flag = left_def
		set_variable = {ROOT.left_def_tee_number = THIS.tee_number}
		set_variable = {ROOT.left_def_name = THIS}
		set_variable = {ROOT.left_def_fb = THIS.fb_power}
		log = "[THIS.GetName] is on left defense"
	}
	random_character = {
		limit = {
			has_character_flag = is_defender
			has_character_flag = is_active
			NOT = {
				has_character_flag = left_def
			}
		}
		set_character_flag = left_def
		set_character_flag = center_def
		set_variable = {ROOT.left_center_def_tee_number = THIS.tee_number}
		set_variable = {ROOT.left_center_def_name = THIS}
		set_variable = {ROOT.left_center_def_fb = THIS.fb_power}
		log = "[THIS.GetName] is on left-center defense"
	}
	random_character = {

			has_character_flag = is_defender
			has_character_flag = is_active
			NOT = {
				has_character_flag = left_def
			}
		}
		set_character_flag = center_def
		set_character_flag = right_def
		set_variable = {ROOT.right_center_def_tee_number = THIS.tee_number}
		set_variable = {ROOT.right_center_def_name = THIS}
		set_variable = {ROOT.right_center_def_fb = THIS.fb_power}
		log = "[THIS.GetName] is on right-center defense"
	}
	random_character = {
		limit = {
			has_character_flag = is_defender
			has_character_flag = is_active
			NOT = {
				has_character_flag = left_def
			}
			NOT = {
				has_character_flag = center_def
			}
			NOT = {
				has_character_flag = right_def
			}

		set_character_flag = right_def
		set_variable = {ROOT.right_def_tee_number = THIS.tee_number}

	random_character = {
		limit = {

		}
		set_variable = {ROOT.gk_tee_number = THIS.tee_number}
		set_variable = {ROOT.gk_name = THIS}
		set_variable = {ROOT.gk_fb = THIS.fb_power}
		log = "[THIS.GetName] is goalkeeper"
	}
	BRZ_calculate_stats = yes
}

BRZ_rebuild_formation = {
	clear_team_ui_flags = yes

	# Remember who was already on the field
	every_character = {
		limit = {
			has_character_flag = is_footballer
			has_character_flag = is_active
		}
		set_character_flag = was_active_before_rebuild
	}

	# Fully clear the old formation
	every_character = {
		limit = {
			has_character_flag = is_footballer
		}

		clr_character_flag = is_active

		clr_character_flag = left_atk
		clr_character_flag = center_atk
		clr_character_flag = right_atk

		clr_character_flag = left_mid
		clr_character_flag = center_mid
		clr_character_flag = right_mid
		clr_character_flag = left_center_mid
		clr_character_flag = right_center_mid

		clr_character_flag = left_def
		clr_character_flag = center_def
		clr_character_flag = right_def
		clr_character_flag = left_center_def
		clr_character_flag = right_center_def

		clr_character_flag = gk
	}

	BRZ_clear_formation_display_vars = yes

	# Goalkeeper always exists
	BRZ_assign_gk = yes

	if = {
		limit = {
			check_variable = { current_team_setup = 442 }
		}

		BRZ_assign_442_slots = yes
	}
	else_if = {
		limit = {
			check_variable = { current_team_setup = 433 }
		}

		BRZ_assign_433_slots = yes
	}
	else_if = {
		limit = {
			check_variable = { current_team_setup = 343 }
		}

		BRZ_assign_343_slots = yes
	}

	# Cleanup temporary memory
	every_character = {
		limit = {
			has_character_flag = was_active_before_rebuild
		}
		clr_character_flag = was_active_before_rebuild
	}

	BRZ_calculate_stats = yes
}

BRZ_assign_gk = {
	if = {
		limit = {
			any_character = {
				has_character_flag = is_footballer
				has_character_flag = is_goalkeeper
				has_character_flag = was_active_before_rebuild
				not = { has_character_flag = is_active }
			}
		}

		random_character = {
			limit = {
				has_character_flag = is_footballer
				has_character_flag = is_goalkeeper
				has_character_flag = was_active_before_rebuild
				not = { has_character_flag = is_active }
			}

			set_character_flag = is_active
			set_character_flag = gk

			set_variable = { ROOT.gk_name = THIS }
			set_variable = { ROOT.gk_tee_number = THIS.tee_number }
			set_variable = { ROOT.gk_fb = THIS.fb_power }

			log = "[THIS.GetName] assigned as goalkeeper"
		}
	}
	else = {
		random_character = {
			limit = {
				has_character_flag = is_footballer
				has_character_flag = is_goalkeeper
				not = { has_character_flag = is_active }
			}

			set_character_flag = is_active
			set_character_flag = gk

			set_variable = { ROOT.gk_name = THIS }
			set_variable = { ROOT.gk_tee_number = THIS.tee_number }
			set_variable = { ROOT.gk_fb = THIS.fb_power }

			log = "[THIS.GetName] assigned as goalkeeper"
		}
	}
}

BRZ_calculate_stats = {
	set_temp_variable = {
		attacker_count = 0
	}
	set_temp_variable = {
		midfielder_count = 0
	}
	set_temp_variable = {
		defender_count = 0
	}
	set_variable = {
		zwkp_atk = 0
	}
	set_variable = {
		zwkp_mid = 0
	}
	set_variable = {
		zwkp_def = 0
	}
	every_character = {
		limit = {
			has_character_flag = is_active
			has_character_flag = is_attacker
		}
		add_to_temp_variable = {
			ROOT.attacker_count = 1
		}
		add_to_variable = {
			ROOT.zwkp_atk = this.fb_power
		}
	}
	every_character = {
		limit = {
			has_character_flag = is_active
			has_character_flag = is_midfielder
		}
		add_to_temp_variable = {
			ROOT.midfielder_count = 1
		}
		add_to_variable = {
			ROOT.zwkp_mid = this.fb_power
		}
	}
	every_character = {
		limit = {
			has_character_flag = is_active
			or = {
				has_character_flag = is_defender
				has_character_flag = is_goalkeeper
			}
		}
		add_to_temp_variable = {
			ROOT.defender_count = 1
		}
		add_to_variable = {
			ROOT.zwkp_def = this.fb_power
		}
	}
	log = "[?defender_count]"
	if = {
		limit = {
			check_variable = {
				attacker_count = 3
			}
		}
		multiply_variable = {
			zwkp_atk = 1.1
		}
	}
	if = {
		limit = {
			check_variable = {
				midfielder_count = 3
			}
		}
		multiply_variable = {
			zwkp_mid = 0.95
		}
	}
	if = {
		limit = {
			check_variable = {
				defender_count = 4
			}
		}
		multiply_variable = {
			zwkp_def = 0.95
		}
	}
	if = {
		limit = {
			check_variable = {
				team_stance = 1
			}
		}
		multiply_variable = {
			zwkp_atk = 1.1
		}
		multiply_variable = {
			zwkp_def = 0.9
		}
	}
	if = {
		limit = {
			check_variable = {
				team_stance = -1
			}
		}
		multiply_variable = {
			zwkp_atk = 0.9
		}
		multiply_variable = {
			zwkp_def = 1.1
		}
	}
	divide_variable = {
		zwkp_atk = attacker_count
	}
	divide_variable = {
		zwkp_mid = midfielder_count
	}
	divide_variable = {
		zwkp_def = defender_count
	}
	set_temp_variable = {
		zwkp_atk_bonus = 1
	}
	add_to_temp_variable = {
		zwkp_atk_bonus = modifier@football_team_attack_factor
	}
	multiply_variable = {
		zwkp_atk = zwkp_atk_bonus
	}
	set_temp_variable = {
		zwkp_mid_bonus = 1
	}
	add_to_temp_variable = {
		zwkp_mid_bonus = modifier@football_team_midfield_factor
	}
	multiply_variable = {
		zwkp_mid = zwkp_mid_bonus
	}
	set_temp_variable = {
		zwkp_def_bonus = 1
	}
	add_to_temp_variable = {
		zwkp_def_bonus = modifier@football_team_defence_factor
	}
	multiply_variable = {
		zwkp_def = zwkp_def_bonus
	}
	round_variable = zwkp_atk
	round_variable = zwkp_mid
	round_variable = zwkp_def
}

league_apply_match_result = {
	add_to_variable = {
		league_goals_shot^league_home = league_home_goals
	}
	add_to_variable = {
		league_goals_lost^league_home = league_away_goals
	}
	add_to_variable = {
		league_goals_shot^league_away = league_away_goals
	}
	add_to_variable = {
		league_goals_lost^league_away = league_home_goals
	}

	if = {
		limit = {
			check_variable = {
				league_home = 0
			}
		}

		set_variable = {
			league_olimpia_last_goals = league_home_goals
		}
		set_variable = {
			league_olimpia_last_goals_against = league_away_goals
		}
		set_variable = {
			league_last_opponent = league_team_tokens^league_away
		}
	}
	else_if = {
		limit = {
			check_variable = {
				league_away = 0
			}
		}

		set_variable = {
			league_olimpia_last_goals = league_away_goals
		}
		set_variable = {
			league_olimpia_last_goals_against = league_home_goals
		}
		set_variable = {
			league_last_opponent = league_team_tokens^league_home
		}
	}

	if = {
		limit = {
			check_variable = {
				league_home_goals = league_away_goals
			}
		}
		add_to_variable = { league_points^league_home = 1 }
		add_to_variable = { league_points^league_away = 1 }
		add_to_variable = { league_draws^league_home = 1 }
		add_to_variable = { league_draws^league_away = 1 }
	}
	else_if = {
		limit = {
			check_variable = {
				league_home_goals > league_away_goals
			}
		}
		add_to_variable = { league_points^league_home = 3 }
		add_to_variable = { league_wins^league_home = 1 }
		add_to_variable = { league_loses^league_away = 1 }
	}
	else_if = {
		limit = {
			check_variable = {
				league_home_goals < league_away_goals
			}
		}
		add_to_variable = { league_points^league_away = 3 }
		add_to_variable = { league_wins^league_away = 1 }
		add_to_variable = { league_loses^league_home = 1 }
	}

	if = {
		limit = {
			OR = {
				check_variable = { league_home = 0 }
				check_variable = { league_away = 0 }
			}
		}

		nsfl_check_corruption_leak = yes
	}
}


league_recalculate_scores = {
	clear_array = league_sort_score

	set_temp_variable = { league_i = 0 }

	while_loop_effect = {
		limit = {
			check_variable = {
				league_i < league_standings^num
			}
		}

		# score = points * 10000000
		set_temp_variable = {
			league_score = league_points^league_i
		}
		multiply_temp_variable = {
			league_score = 1000000
		}

		# score += wins * 100000
		set_temp_variable = {
			league_score_part = league_wins^league_i
		}
		multiply_temp_variable = {
			league_score_part = 10000
		}
		add_to_temp_variable = {
			league_score = league_score_part
		}

		# goal_difference = goals_shot - goals_lost
		set_temp_variable = {
			league_goal_difference = league_goals_shot^league_i
		}
		subtract_from_temp_variable = {
			league_goal_difference = league_goals_lost^league_i
		}

		# score += goal_difference * 100
		multiply_temp_variable = {
			league_goal_difference = 100
		}
		add_to_temp_variable = {
			league_score = league_goal_difference
		}

		# score += goals scored
		add_to_temp_variable = {
			league_score = league_goals_shot^league_i
		}

		add_to_array = {
			array = league_sort_score
			value = league_score
			index = league_i
		}

		add_to_temp_variable = { league_i = 1 }
	}
}

league_sort_standings = {
	league_recalculate_scores = yes

	clear_array = league_standings

	set_temp_variable = { league_pick_count = 0 }

	while_loop_effect = {
		limit = {
			check_variable = {
				league_pick_count < league_team_tokens^num
			}
		}

		find_highest_in_array = {
			array = league_sort_score
			value = league_best_score
			index = league_best_index
		}

		add_to_array = {
			array = league_standings
			value = league_team_tokens^league_best_index
		}

		set_variable = {
			league_sort_score^league_best_index = -999999999
		}

		add_to_temp_variable = { league_pick_count = 1 }
	}
}


nsfl_simulate_generic_match = {
	set_temp_variable = { league_home_goals = 0 }
	set_temp_variable = { league_away_goals = 0 }

	set_temp_variable = {
		league_strength_difference = league_strength^league_home
	}
	subtract_from_temp_variable = {
		league_strength_difference = league_strength^league_away
	}

	# gramy u siebie, hej drużyna gramy u siebie
	add_to_temp_variable = {
		league_strength_difference = 3
	}

	# dzwoń do fryzjera
	nsfl_apply_olimpia_influence = yes

	set_temp_variable = { league_match_phase = 0 }

	while_loop_effect = {
		limit = {
			check_variable = {
				league_match_phase < 5
			}
		}

		nsfl_roll_home_goal = yes
		nsfl_roll_away_goal = yes

		add_to_temp_variable = {
			league_match_phase = 1
		}
	}
}

BRZ_select_local_promoted_club = {
	if = {
		limit = {
			check_variable = {
				league_standings^0 = token:BRZ_olimpia_truskolasy
			}
		}

		random_list = {
			50 = {
				set_variable = {
					BRZ_promoted_local_club = league_standings^1
				}
			}
			50 = {
				set_variable = {
					BRZ_promoted_local_club = league_standings^2
				}
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				check_variable = {
					league_standings^1 = token:BRZ_olimpia_truskolasy
				}
				check_variable = {
					league_standings^2 = token:BRZ_olimpia_truskolasy
				}
			}
		}

		set_variable = {
			BRZ_promoted_local_club = league_standings^0
		}
	}
	else = {
		# Safety fallback
		set_variable = {
			BRZ_promoted_local_club = league_standings^0
		}
	}
}

league_load_current_batch_matches = {
	set_temp_variable = { league_round_start_index = league_round }
	subtract_from_temp_variable = { league_round_start_index = 1 }
	multiply_temp_variable = {
		var = league_round_start_index
		value = league_matches_per_round
	}
	add_to_temp_variable = { league_round_start_index = league_batch_offset }

	set_temp_variable = { league_match_index = league_round_start_index }

	set_variable = { league_match_1_home_index = league_fixture_home_index^league_match_index }
	set_variable = { league_match_1_away_index = league_fixture_away_index^league_match_index }

	set_variable = { league_match_1_home = league_team_tokens^league_match_1_home_index }
	set_variable = { league_match_1_away = league_team_tokens^league_match_1_away_index }

	set_temp_variable = { league_match_index = league_round_start_index }
	add_to_temp_variable = { league_match_index = 1 }

	set_variable = { league_match_2_home_index = league_fixture_home_index^league_match_index }
	set_variable = { league_match_2_away_index = league_fixture_away_index^league_match_index }

	set_variable = { league_match_2_home = league_team_tokens^league_match_2_home_index }
	set_variable = { league_match_2_away = league_team_tokens^league_match_2_away_index }

	set_temp_variable = { league_match_index = league_round_start_index }
	add_to_temp_variable = { league_match_index = 2 }

	set_variable = { league_match_3_home_index = league_fixture_home_index^league_match_index }
	set_variable = { league_match_3_away_index = league_fixture_away_index^league_match_index }

	set_variable = { league_match_3_home = league_team_tokens^league_match_3_home_index }
	set_variable = { league_match_3_away = league_team_tokens^league_match_3_away_index }

	set_temp_variable = { league_match_index = league_round_start_index }
	add_to_temp_variable = { league_match_index = 3 }

	set_variable = { league_match_4_home_index = league_fixture_home_index^league_match_index }
	set_variable = { league_match_4_away_index = league_fixture_away_index^league_match_index }

	set_variable = { league_match_4_home = league_team_tokens^league_match_4_home_index }
	set_variable = { league_match_4_away = league_team_tokens^league_match_4_away_index }

	if = {
		limit = {
			check_variable = { league_matches_this_batch > 4 }
		}

		set_country_flag = league_match_5_available

		set_temp_variable = { league_match_index = league_round_start_index }
		add_to_temp_variable = { league_match_index = 4 }

		set_variable = { league_match_5_home_index = league_fixture_home_index^league_match_index }
		set_variable = { league_match_5_away_index = league_fixture_away_index^league_match_index }

		set_variable = { league_match_5_home = league_team_tokens^league_match_5_home_index }
		set_variable = { league_match_5_away = league_team_tokens^league_match_5_away_index }
	}
	else = {
		clr_country_flag = league_match_5_available
	}
}

	add_to_variable = { league_matches_finished_in_batch = 1 }

	if = {
		limit = {
			check_variable = { league_batch = 1 }
			check_variable = {
				league_matches_finished_in_batch = 3
			}
		}

		country_event = {
			id = brz_league.5
			days = 3
		}
	}

		limit = {

			check_variable = {
				league_matches_finished_in_batch = 4
			}
		}

		league_finish_voivodeship_round = yes
	}
}
