coring_puppet_lodzkie = {
	every_owned_state = {
		limit = {
			is_lodz_state = yes
			NOT = { is_core_of = ROOT }
			is_owned_and_controlled_by = ROOT
		}
		LDZ = { transfer_state = PREV }
		PaF_change_capital_coring_prequel = yes
	}
	LDZ = { transfer_state = 580 }
	LDZ = { get_current_government_type = yes }
	ROOT = {
		diplomatic_relation = {
			country = LDZ
			relation = puppet
			active = yes
		}
	}
	LDZ = { restore_previous_government_type = yes }	
	ROOT = {
		set_autonomy = {
		    target = LDZ
		    autonomous_state = autonomy_colony
		}
	}
	LDZ = {
		meta_effect = {	
			text = {
				set_cosmetic_tag = LDZ_[OWNER_TAG]
			}
			OWNER_TAG = "[?ROOT.GetTag]"
		}
		meta_effect = {	
			text = {
				set_party_name = {
					ideology = ROOT
					long_name = [OWNER_TAG]_[OWNER_IDEOLOGY]_party_long
					name = [OWNER_TAG]_[OWNER_IDEOLOGY]_party
				}
			}
			OWNER_TAG = "[?ROOT.GetTag]"
			OWNER_IDEOLOGY = "[?ROOT.GetRulingIdeology]"
		}
		meta_effect = {	
			text = {
				add_country_leader_role = {
				    character = LDZ_[OWNER_TAG]_leader
				    promote_leader = yes
				    country_leader = {
				        ideology = [OWNER_IDEOLOGY]_subtype
				    }
				}
			}
			OWNER_TAG = "[?ROOT.GetTag]"
			OWNER_IDEOLOGY = "[?ROOT.GetRulingIdeology]"
		}
		add_ideas = trade_colonial
		custom_effect_tooltip = rozdzielacz_efektow
		add_ideas = coring_puppet_idea
		custom_effect_tooltip = rozdzielacz_efektow
		remove_ideas = podlasie_tribe_idea		
		inherit_technology = ROOT	
	}
}
