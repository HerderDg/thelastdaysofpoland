scripted_gui = {
	culture_window = {
		context_type = selected_state_context
		ai_enabled = {
			always = no
		}
		window_name = "culture_icon_container"
		parent_window_token = selected_state_view

		properties = {
			culture_icon = {
				image = "GFX_culture_[?subculture]"
			}	
			culture_change_icon = {
				image = "[GetCurrentCultureChangerIcon]"
			}
		}
		triggers = {
			culture_change_icon_bg_visible = {
				THIS = {
					is_owned_by = ROOT
				}
			}
			culture_change_icon_visible = {
				THIS = {
					is_owned_by = ROOT
				}
			}							
			culture_change_icon_bg_click_enabled = {
				THIS = {
					NOT = {
						check_variable = {
							subculture = ROOT.official_culture
						}
					}
				}
				ROOT = {
					NOT = { 
						any_owned_state = {
							has_state_flag = state_culture_conversion_in_progress
						}
					}						
				}				
			}			
		}
		effects = {
			culture_change_icon_bg_click = {
				THIS = {
					set_state_flag = state_culture_conversion_in_progress
				}
				ROOT = { 
					country_event = {
						id = config.8
					}
				}				
			}
		}
	}

	open_culture_gui = {
		context_type = player_context
		parent_window_token = politics_tab
		window_name = "culture_button_container"
		properties = {
			show_culture_gui = {
				image = "[GetCurrentCultureIcon]"
			}
		}
		ai_enabled = {
			always = no
		}
		effects = {
			show_culture_gui_click = {
				if = {
					limit = { has_variable = global.PaF_CultureGUI }
					clear_variable = global.PaF_CultureGUI
				}
				else_if = {
					limit = { has_variable = global.PaF_ReligionGUI }
					clear_variable = global.PaF_ReligionGUI
					set_variable = { global.PaF_CultureGUI = 1 }
				}
				else = {
					set_variable = { global.PaF_CultureGUI = 1 }
				}	
			}
		}
	}

	culture_gui = {
		window_name = "culture_container_gui"
		context_type = player_context
		visible = {
			has_variable = global.PaF_CultureGUI
		}
		ai_enabled = {
			always = no
		}
		properties = {
			current_culture_icon = {
				image = "[GetCurrentCultureIcon]"
			}
			dominant_culture_icon = {
				image = "[GetDominantCultureIcon]"
			}
			cultural_law_value = {
				image = "[GetCurrentCultureLawIcon]"
			}
		}
		effects = {
			cultural_law_close_click = {
				if = {
					limit = { has_variable = global.PaF_CultureGUI }
					clear_variable = global.PaF_CultureGUI
				}
				else = {
					set_variable = { global.PaF_CultureGUI = 1 }
				}
			}
		}
	}


}
